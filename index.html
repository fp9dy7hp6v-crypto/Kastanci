        function activateTab(tabName) {

            // Hide all contents

            document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));

            // Show target

            const target = document.getElementById('content-' + tabName);

            if (target) target.classList.add('active');


            // Style buttons

            document.querySelectorAll('.tab-btn').forEach(btn => {

                if (btn.getAttribute('data-tab') === tabName) {

                    btn.classList.add('border-brand-500', 'text-brand-500');

                    btn.classList.remove('border-transparent', 'text-slate-400');

                } else {

                    btn.classList.remove('border-brand-500', 'text-brand-500');

                    btn.classList.add('border-transparent', 'text-slate-400');

                }

            });


            document.querySelectorAll('.mobile-nav-btn').forEach(btn => {

                if (btn.getAttribute('data-tab') === tabName) {

                    btn.classList.add('text-brand-500');

                    btn.classList.remove('text-slate-400');

                } else {

                    btn.classList.remove('text-brand-500');

                    btn.classList.add('text-slate-400');

                }

            });

        }


        // Add robust click listeners

        document.addEventListener('click', function(e) {

            const btn = e.target.closest('[data-tab]');

            if (btn) {

                e.preventDefault();

                const tabName = btn.getAttribute('data-tab');

                activateTab(tabName);

            }

        });

        function filterPlayers() {

            const searchEl = document.getElementById('playerSearch');

            const sortEl = document.getElementById('sortBy');

            const query = searchEl ? searchEl.value.toLowerCase() : '';

            const sortBy = sortEl ? sortEl.value : 'points';


            let filtered = playersData.filter(p => p.name.toLowerCase().includes(query));


            filtered.sort((a, b) => {

                if(sortBy === 'goals') return b.goals - a.goals || b.points - a.points;

                if(sortBy === 'assists') return b.assists - a.assists || b.points - a.points;

                if(sortBy === 'points') return b.points - a.points || b.goals - a.goals || b.plusMinus - a.plusMinus;

                if(sortBy === 'plusMinus') return b.plusMinus - a.plusMinus || b.points - a.points;

                if(sortBy === 'matches') return b.matches - a.matches || b.points - a.points;

                return 0;

            });


            renderPlayers(filtered);

        }
